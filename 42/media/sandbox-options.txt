VERSION = 1,

/*
 * NPC Crew -- bear cubs (B1a: find and tame a cub). Page BanditsNPC ("NPC Crew").
 * Every value is read by BanditsNPC.Bear.Opt with a fallback to the default below; in multiplayer the server's values count.
 * Kevin 12.11.2026: from 39 options exactly twelve stay; all others are fixed values in BanditsNPC.Bear.CFG (same numbers as the old defaults).
 * A save with values set for removed options still loads: those values are never read.
 */

/* Main switch: off = no new cubs (forest and bait); cubs already in the world stay tameable. */
option BanditsNPC.BearCubsEnabled
{
	type = boolean, default = true,
	page = BanditsNPC,
	translation = BanditsNPC_BearCubsEnabled,
}

/* 1 off, 2 rare (2 % per in-game hour), 3 normal (5 %), 4 often (12 %), 5 very often (30 %); rolled every 10 in-game minutes with a sixth of that. */
option BanditsNPC.BearCubForestRate
{
	type = enum, numValues = 5, default = 3,
	page = BanditsNPC,
	translation = BanditsNPC_BearCubForestRate,
	valueTranslation = BanditsNPC_BearCubForestRate,
}

/* Tamed bears per player (Kevin 08.11.2026: one). */
option BanditsNPC.BearMaxPerPlayer
{
	type = integer, min = 1, max = 4, default = 1,
	page = BanditsNPC,
	translation = BanditsNPC_BearMaxPerPlayer,
}

/* Bait spot (honey or fish) on or off. */
option BanditsNPC.BearBaitEnabled
{
	type = boolean, default = true,
	page = BanditsNPC,
	translation = BanditsNPC_BearBaitEnabled,
}

/* Taming speed in percent: points per feeding and petting (100 = normal: 4 per feeding, 3 per petting). The minimum of 3 feedings and the waits between feedings (2 in-game hours) and pettings (1 in-game hour) stay. Replaces the four taming options (Kevin 12.11.2026). */
option BanditsNPC.BearTamePercent
{
	type = integer, min = 25, max = 1000, default = 100,
	page = BanditsNPC,
	translation = BanditsNPC_BearTamePercent,
}

/* In-game days from taming until a cub is fully grown, size 0.35 -> 1.35 (Kevin 11.11.2026: 100). Faster at first: at 100 days size 0.75 on day 15 and 1.0 on day 45; other values scale the same way. */
option BanditsNPC.BearGrowDays
{
	type = integer, min = 10, max = 365, default = 100,
	page = BanditsNPC,
	translation = BanditsNPC_BearGrowDays,
}

/* Off (default): a tamed bear can die. On: it goes down instead; helped up with 45 percent, alone after 12 in-game hours with 15 percent (Kevin 08.11.2026). */
option BanditsNPC.BearCannotDie
{
	type = boolean, default = false,
	page = BanditsNPC,
	translation = BanditsNPC_BearCannotDie,
}

/* A tamed bear stops fighting and comes back when its health lasts only this many more hits of its current foes (Kevin 12.11.2026: 2-3). 0 = never. */
option BanditsNPC.BearFleeHits
{
	type = integer, min = 0, max = 10, default = 3,
	page = BanditsNPC,
	translation = BanditsNPC_BearFleeHits,
}

/* Damage a tamed bear deals, on top of growing and Fighting (100 = normal). */
option BanditsNPC.BearDamagePercent
{
	type = integer, min = 10, max = 300, default = 100,
	page = BanditsNPC,
	translation = BanditsNPC_BearDamagePercent,
}

/* Healing speed in percent of a tamed bear on its own (100 = normal: about 1 percent of its health per in-game hour; 0 = never on its own). Bandages are not affected. Replaces the two healing options (Kevin 12.11.2026); bears do not sleep yet. */
option BanditsNPC.BearHealPercent
{
	type = integer, min = 0, max = 1000, default = 100,
	page = BanditsNPC,
	translation = BanditsNPC_BearHealPercent,
}

/* Skill learning speed in percent; scales every credit and the per-minute cap (Kevin 12.11.2026: slower by default, faster by sandbox). */
option BanditsNPC.BearSkillXpPercent
{
	type = integer, min = 25, max = 1000, default = 100,
	page = BanditsNPC,
	translation = BanditsNPC_BearSkillXpPercent,
}

/* Skills limited by growth stage: cub 3, half-grown 6, strong 8, full-grown 10 (Kevin 12.11.2026). */
option BanditsNPC.BearSkillAgeCap
{
	type = boolean, default = true,
	page = BanditsNPC,
	translation = BanditsNPC_BearSkillAgeCap,
}
